1. Executive Summary
This document is the complete Product Requirements Document (PRD) for the Hedge Fund Risk Modeling & Semi-Automated Trading System, built as part of Code2Create Challenge Round 3. It serves as the single source of truth for the team — covering architecture decisions, dataset analysis, module breakdown, implementation order, and sprint timeline.

The system ingests four financial datasets (equity, oil, multi-asset, macroeconomic) spanning 2020-2047 with 10,000 daily records each. It runs a full backtesting pipeline: preprocessing, feature engineering, signal generation, risk modeling, position sizing, simulation, and a visual dashboard.

Core Goal
Maximize risk-adjusted returns (Sharpe Ratio) while minimizing drawdown across a multi-asset portfolio
Strategy Type
Rule-based semi-automated (momentum + volatility signals) with optional ML overlay
Key Deliverable
Interactive Streamlit dashboard showing portfolio performance, risk metrics, and trade logs

4.2 data/loader.py — Data Ingestion (Issues 1, 16)
Loads all 4 CSVs, validates schema, merges on Date. Returns a single merged DataFrame with all features available for downstream use.
import pandas as pd

REQUIRED_COLS = {
    "equity": ["Date","Price","Volume","Returns","SMA_10"],
    "oil": ["Date","Price","Volume","Returns","Volatility"],
    "multi": ["Date","Oil","Gold","Bonds","Oil_Returns","Gold_Returns"],
    "macro": ["Date","Inflation","Interest_Rate","USD_Index","Sentiment"]
}

def load_all(data_dir="datasets/"):
    dfs = {}
    for key, cols in REQUIRED_COLS.items():
        path = f"{data_dir}/{key}_dataset.csv"
        df = pd.read_csv(path, parse_dates=["Date"])
        # Issue 16: schema validation
        missing = set(cols) - set(df.columns)
        if missing:
            raise ValueError(f"Missing columns in {key}: {missing}")
        df = df.rename(columns={"Price": f"{key}_Price"})  # avoid collisions
        dfs[key] = df.set_index("Date")
    merged = dfs["equity"].join([dfs["oil"], dfs["multi"], dfs["macro"]], how="inner")
    return merged.reset_index()

4.3 data/preprocessor.py — Cleaning & Validation (Issue 2)
Handles NaN imputation and outlier smoothing. CRITICAL: use forward-fill only (no future data leakage). Clip extreme values at ±3 standard deviations.
import numpy as np

def preprocess(df):
    df = df.copy().sort_values("Date").reset_index(drop=True)
    
    # 1. Forward-fill NaNs (no look-ahead bias)
    df.ffill(inplace=True)
    df.bfill(inplace=True)  # only for initial rows with no prior data
    
    # 2. Clip outliers at ±3 std per numeric column
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    for col in numeric_cols:
        mean, std = df[col].mean(), df[col].std()
        df[col] = df[col].clip(mean - 3*std, mean + 3*std)
    
    # 3. Remove duplicate dates
    df = df.drop_duplicates(subset="Date")
    
    return df

4.4 features/engineer.py — Feature Engineering (Issues 3, 4)
Computes rolling volatility, momentum, RSI-style momentum, and aligns macro signals. All rolling windows use .shift(1) to prevent look-ahead.
from config import LOOKBACK_VOL, MOMENTUM_WINDOW

def engineer_features(df):
    df = df.copy()
    
    # ── Equity Features ──
    df["eq_volatility"] = df["equity_Returns"].rolling(LOOKBACK_VOL).std() * (252**0.5)
    df["eq_momentum"]   = df["equity_Price"].pct_change(MOMENTUM_WINDOW).shift(1)
    df["eq_SMA_ratio"]  = df["equity_Price"] / df["SMA_10"] - 1   # price vs SMA
    
    # ── Oil Features ──
    df["oil_momentum"]  = df["oil_Returns"].rolling(MOMENTUM_WINDOW).mean().shift(1)
    # use pre-computed volatility from dataset directly
    df["oil_vol"]       = df["Volatility"].shift(1)
    
    # ── Gold / Bonds Features (from multi_asset) ──
    df["gold_momentum"] = df["Gold_Returns"].rolling(MOMENTUM_WINDOW).mean().shift(1)
    df["bond_trend"]    = df["Bonds"].pct_change(MOMENTUM_WINDOW).shift(1)
    
    # ── Macro Alignment (already daily, just normalize) ──
    for col in ["Inflation","Interest_Rate","USD_Index","Sentiment"]:
        df[f"macro_{col}"] = (df[col] - df[col].rolling(60).mean()) / df[col].rolling(60).std()
    
    df.dropna(inplace=True)  # drop initial window rows
    return df


4.5 strategy/signals.py — Signal Generation (Issue 8)
Rule-based engine. For each asset on each date: momentum > threshold AND volatility < cap = BUY; momentum < -threshold = SELL; else HOLD. Macro sentiment acts as a global filter.
from config import SIGNAL_THRESHOLD

def generate_signals(df):
    signals = df[["Date"]].copy()
    
    # ── Equity Signal ──
    cond_buy  = (df["eq_momentum"] > SIGNAL_THRESHOLD) & (df["eq_volatility"] < 0.4)
    cond_sell = (df["eq_momentum"] < -SIGNAL_THRESHOLD)
    signals["Equity"] = "HOLD"
    signals.loc[cond_buy, "Equity"]  = "BUY"
    signals.loc[cond_sell, "Equity"] = "SELL"
    
    # ── Oil Signal ──
    cond_oil_buy  = (df["oil_momentum"] > SIGNAL_THRESHOLD) & (df["oil_vol"] < 0.5)
    cond_oil_sell = (df["oil_momentum"] < -SIGNAL_THRESHOLD)
    signals["Oil"] = "HOLD"
    signals.loc[cond_oil_buy,  "Oil"] = "BUY"
    signals.loc[cond_oil_sell, "Oil"] = "SELL"
    
    # ── Gold Signal (safe-haven: buy when macro_Sentiment < -0.5) ──
    signals["Gold"] = "HOLD"
    signals.loc[df["macro_Sentiment"] < -0.5, "Gold"] = "BUY"
    signals.loc[df["gold_momentum"] > 0.03,   "Gold"] = "SELL"
    
    # ── Bonds Signal ──
    signals["Bonds"] = "HOLD"
    signals.loc[(df["macro_Interest_Rate"] < -0.3) & (df["bond_trend"] > 0), "Bonds"] = "BUY"
    signals.loc[df["macro_Interest_Rate"] > 0.5, "Bonds"] = "SELL"
    
    return signals

4.6 portfolio/state.py — Portfolio State (Issue 5)
Maintains the live state of the portfolio: cash, positions per asset, total value. Updated after every trade and after daily mark-to-market.
from config import INITIAL_CAPITAL, MAX_POSITION_PCT

class PortfolioState:
    def __init__(self):
        self.cash = INITIAL_CAPITAL
        self.positions = {"Equity": 0, "Oil": 0, "Gold": 0, "Bonds": 0}
        self.history = []  # daily snapshots
    
    def total_value(self, prices: dict) -> float:
        return self.cash + sum(qty * prices[a] for a, qty in self.positions.items())
    
    def max_position_value(self, prices):
        return self.total_value(prices) * MAX_POSITION_PCT
    
    def snapshot(self, date, prices):
        self.history.append({
            "date": date,
            "total_value": self.total_value(prices),
            "cash": self.cash,
            **{f"pos_{k}": v for k, v in self.positions.items()}
        })

4.7 portfolio/sizer.py — Position Sizing (Issue 9)
Volatility-adjusted sizing. Higher volatility = smaller position. Uses a target-risk approach: allocate capital such that the dollar risk per trade is constant.
from config import MAX_POSITION_PCT

def compute_position_size(asset, portfolio_value, asset_price, asset_volatility, risk_pct=0.01):
    """
    risk_pct: % of portfolio risked per trade (default 1%)
    dollar_risk = portfolio_value * risk_pct
    shares = dollar_risk / (asset_price * asset_volatility)
    capped at MAX_POSITION_PCT of portfolio
    """
    if asset_volatility <= 0 or asset_price <= 0:
        return 0
    dollar_risk = portfolio_value * risk_pct
    shares = dollar_risk / (asset_price * asset_volatility)
    max_shares = (portfolio_value * MAX_POSITION_PCT) / asset_price
    return min(shares, max_shares)


4.8 portfolio/simulator.py — Trade Simulation (Issues 10, 11, 15)
The main backtest loop. Iterates daily, reads signals, executes trades with friction costs, triggers monthly rebalancing, and records portfolio snapshots. Handles insufficient capital gracefully.
from config import TRANSACTION_COST, SLIPPAGE, REBALANCE_FREQ

def simulate(df_features, signals, state, logger):
    prices_cols = {"Equity":"equity_Price","Oil":"oil_Price","Gold":"Gold","Bonds":"Bonds"}
    vol_cols    = {"Equity":"eq_volatility","Oil":"oil_vol","Gold":"eq_volatility","Bonds":"eq_volatility"}
    
    for i, row in df_features.iterrows():
        prices = {a: row[c] for a, c in prices_cols.items()}
        state.snapshot(row["Date"], prices)
        sig_row = signals.loc[signals["Date"] == row["Date"]].iloc[0]
        portfolio_val = state.total_value(prices)
        
        for asset in ["Equity","Oil","Gold","Bonds"]:
            signal = sig_row[asset]
            price  = prices[asset]
            vol    = row.get(vol_cols[asset], 0.2)
            
            if signal == "BUY" and state.positions[asset] == 0:
                qty = compute_position_size(asset, portfolio_val, price, vol)
                cost = qty * price * (1 + TRANSACTION_COST + SLIPPAGE)
                if cost > state.cash:          # Issue 15: insufficient capital
                    logger.log_error(row["Date"], asset, "INSUFFICIENT_CAPITAL", cost, state.cash)
                    continue
                state.cash -= cost
                state.positions[asset] += qty
                logger.log_trade(row["Date"], asset, "BUY", qty, price, vol, sig_row)
            
            elif signal == "SELL" and state.positions[asset] > 0:
                qty = state.positions[asset]
                proceeds = qty * price * (1 - TRANSACTION_COST - SLIPPAGE)
                state.cash += proceeds
                state.positions[asset] = 0
                logger.log_trade(row["Date"], asset, "SELL", qty, price, vol, sig_row)
        
        # Monthly rebalancing
        if i % REBALANCE_FREQ == 0:
            rebalance(state, prices, logger, row["Date"])
    
    return state

4.9 metrics/calculator.py — Risk Metrics (Issues 6, 7, 12, 13)
Computes all evaluation metrics from the portfolio history DataFrame. Call once after simulation completes.
import numpy as np
from config import RISK_FREE_RATE, VAR_CONFIDENCE

def compute_metrics(history_df, market_returns):
    pv = history_df["total_value"].values
    daily_returns = np.diff(pv) / pv[:-1]
    
    # ── Sharpe Ratio ──
    excess = daily_returns - RISK_FREE_RATE / 252
    sharpe = (excess.mean() / excess.std()) * np.sqrt(252) if excess.std() > 0 else 0
    
    # ── Max Drawdown ──
    roll_max = np.maximum.accumulate(pv)
    drawdowns = (pv - roll_max) / roll_max
    max_drawdown = drawdowns.min()
    
    # ── Annualized Volatility ──
    annualized_vol = daily_returns.std() * np.sqrt(252)
    
    # ── VaR (Historical) ──
    var = np.percentile(daily_returns, (1 - VAR_CONFIDENCE) * 100)
    
    # ── Alpha & Beta ──
    mkt = np.array(market_returns[:len(daily_returns)])
    beta = np.cov(daily_returns, mkt)[0,1] / np.var(mkt) if np.var(mkt) > 0 else 0
    alpha = (daily_returns.mean() - beta * mkt.mean()) * 252
    
    # ── Cumulative & Annualized Return ──
    total_return = (pv[-1] - pv[0]) / pv[0]
    n_years = len(pv) / 252
    annualized_return = (1 + total_return) ** (1 / n_years) - 1
    
    return {"sharpe": round(sharpe,4), "max_drawdown": round(max_drawdown,4),
            "volatility": round(annualized_vol,4), "var_95": round(var,4),
            "alpha": round(alpha,4), "beta": round(beta,4),
            "total_return": round(total_return,4), "annualized_return": round(annualized_return,4)}


4.10 logs/trade_logger.py — Explainable Logs (Issues 14, 15)
Logs every trade decision with the indicator values that triggered it. Also captures errors. Outputs a CSV and is used by the dashboard.
import pandas as pd

class TradeLogger:
    def __init__(self):
        self.trades = []
        self.errors = []
    
    def log_trade(self, date, asset, action, qty, price, volatility, signal_row):
        self.trades.append({
            "date": date, "asset": asset, "action": action,
            "quantity": round(qty, 2), "price": round(price, 4),
            "volatility": round(volatility, 4),
            "rationale": f"{action} triggered: momentum={signal_row.get('eq_momentum','N/A'):.4f}, vol={volatility:.4f}"
        })
    
    def log_error(self, date, asset, error_type, required, available):
        self.errors.append({
            "date": date, "asset": asset, "error": error_type,
            "required_cash": round(required, 2), "available_cash": round(available, 2)
        })
    
    def to_dataframe(self):
        return pd.DataFrame(self.trades)
    
    def save(self, path="trade_log.csv"):
        self.to_dataframe().to_csv(path, index=False)
